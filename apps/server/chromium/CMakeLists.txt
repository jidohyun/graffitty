cmake_minimum_required(VERSION 3.21)
project(GraffittyChromium LANGUAGES C CXX)
if(APPLE)
  message(FATAL_ERROR "Use build.sh on macOS")
endif()
set(CMAKE_CXX_STANDARD 20)
if(WIN32)
  set(USE_SANDBOX ON CACHE BOOL "Require CEF's sandbox bootstrap" FORCE)
else()
  set(USE_SANDBOX OFF CACHE BOOL "Platform sandbox setup" FORCE)
endif()
set(CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
ADD_LOGICAL_TARGET("libcef_lib" "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
if(WIN32)
  # CEF 138+ owns sandbox initialization in its bootstrap executable. Our
  # browser and subprocess entry point lives in the matching client DLL.
  add_library(graffitty_chromium SHARED host_windows.cpp)
  SET_LIBRARY_TARGET_PROPERTIES(graffitty_chromium)
  if(NOT EXISTS "${CEF_BOOTSTRAP_EXE}")
    message(FATAL_ERROR "CEF_BOOTSTRAP_EXE must name the bootstrap from the same fork build as libcef.dll")
  endif()
  add_custom_command(TARGET graffitty_chromium POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${CEF_BOOTSTRAP_EXE}"
      "$<TARGET_FILE_DIR:graffitty_chromium>/graffitty_chromium.exe"
    VERBATIM)
  set(CEF_TARGET_OUT_DIR "$<TARGET_FILE_DIR:graffitty_chromium>")
  SET_LPAC_ACLS(graffitty_chromium)
else()
  find_package(Vulkan REQUIRED)
  add_executable(graffitty_chromium host_linux.cpp)
  SET_EXECUTABLE_TARGET_PROPERTIES(graffitty_chromium)
  target_link_libraries(graffitty_chromium PRIVATE Vulkan::Vulkan)
  set_target_properties(graffitty_chromium PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  # CEF disables exceptions by default; the Vulkan RAII helpers unwind imports
  # and command buffers on driver errors before returning to the CEF callback.
  target_compile_options(graffitty_chromium PRIVATE -fexceptions)
  add_executable(chromium_display_test EXCLUDE_FROM_ALL ../../../tests/native/linux_display_platform_test.cpp)
  target_compile_options(chromium_display_test PRIVATE -UNDEBUG)
  add_library(chromium_capture_test SHARED EXCLUDE_FROM_ALL ../../../tests/native/linux_chromium_capture_test.cpp)
  target_link_libraries(chromium_capture_test PRIVATE Vulkan::Vulkan)
  target_compile_options(chromium_capture_test PRIVATE -fexceptions)
  find_package(Threads REQUIRED)
  add_executable(chromium_pipe_test EXCLUDE_FROM_ALL ../../../tests/native/linux_chromium_control_test.cpp)
  target_link_libraries(chromium_pipe_test PRIVATE Threads::Threads)
endif()
target_include_directories(graffitty_chromium PRIVATE "${CEF_ROOT}")
target_link_libraries(graffitty_chromium PRIVATE libcef_lib libcef_dll_wrapper ${CEF_STANDARD_LIBS})
if(WIN32)
  target_link_libraries(graffitty_chromium PRIVATE d3d11 dxgi comctl32)
  add_executable(chromium_pipe_test EXCLUDE_FROM_ALL ../../../tests/native/windows_chromium_pipe_test.cpp)
  target_link_libraries(chromium_pipe_test PRIVATE ole32 advapi32)
  add_library(chromium_capture_test SHARED EXCLUDE_FROM_ALL
    ../../../crates/transport/chromium-video/src/shared_windows.cpp)
  set_target_properties(chromium_capture_test PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
  target_link_libraries(chromium_capture_test PRIVATE d3d11 d3d12 dxgi bcrypt advapi32)
endif()
